\section{Introduction}
% ============================================================

\begin{frame}{Introduction}

  \textbf{What we have seen so far:}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Bootstrap as a tool to \alert{compute the variance / distribution} of
      a statistic when analytical formulas are unavailable.
    \item Bootstrap for \alert{asymptotic refinements}: faster convergence than the
      first-order normal approximation (Beran 1987, Hall 1992).
  \end{itemize}

  \medskip

  \textbf{Today --- a different question:}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item What if the estimator carries an \alert{asymptotic bias} that the
      bootstrap simply \textbf{cannot} replicate?
    \item Then standard bootstrap inference is \textbf{invalid}, even
      asymptotically.
    \item Cavaliere et al.\ (2024): valid inference can still be restored
      \alert{without estimating the bias}, by transforming the bootstrap
      $p$-value (\textit{prepivoting}, Beran 1987, 1988).
  \end{itemize}

\end{frame}

% ============================================================

\begin{frame}{Setup --- the original statistic (1.1)}

  Scalar parameter $\theta$, estimator $\thn$ with rate $g(n)\to\infty$:
  \[
    \Tn := g(n)(\thn - \theta) \;\dto\; B + \xi_1,
    \tag{1.1}
  \]
  with $\xi_1$ centred at zero, continuous cdf $G$.

  \medskip

  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item $B$: \alert{asymptotic bias}; in the cases of interest \textbf{not consistently estimable}.
    \item Typical case: $g(n) = n^{1/2}$, $\xi_1 \sim N(0,\sigma^2)$.
    \item \textbf{Why classical inference fails:} quantiles of $\xi_1$ ignore $B$.
  \end{itemize}

  \medskip

  \begin{block}{Why bootstrap, not analytical bias correction?}
    A consistent estimator of $B$ often does not exist (model averaging, ridge),
    or exists but performs poorly in samples (nonparametric $\hat\beta''(x)$).
    Hall (1992), Horowitz (2001), Cattaneo et al.\ (2018, 2022): the standard
    bootstrap also \textit{cannot} fix the bias when $B$ cannot be estimated.
  \end{block}

\end{frame}

% ============================================================

\begin{frame}{Setup --- the bootstrap statistic (1.2)}

  Bootstrap statistic --- centred at $\thn$ (the bootstrap world's ``truth''):
  \[
    \Tns := g(n)(\thns - \thn),
    \qquad
    \Tns - \Bnh \;\dstarto_p\; \xi_1.
    \tag{1.2}
  \]

  \medskip

  \begin{itemize}
    \setlength{\itemsep}{8pt}
    \item $\Bnh$: \alert{implicit} bootstrap bias --- what the bootstrap DGP \textit{produces},
      not something the researcher chooses.
    \item Noise $\xi_1$ is replicated correctly under $\Pstar$.
    \item If $\Bnh - B = o_p(1)$, bootstrap is valid in the usual sense $(\ast)$.
  \end{itemize}

  \medskip

  \begin{block}{Notation (used throughout)}
    \footnotesize
    $Y_n^* \pto_p 0$ means $\Pstar(|Y_n^*|>\varepsilon)\pto 0$ for every $\varepsilon>0$.\\
    $Y_n^* \dstarto_p \xi$ means $\Pstar(Y_n^*\le u)\pto P(\xi\le u)$ at every continuity point.
  \end{block}

\end{frame}

% ============================================================

\begin{frame}{The failure mode (1.3)}

  \[
    \Bnh - B \;\dto\; \xi_2 \;\not\equiv\; 0,
    \qquad \text{jointly with (1.1).}
    \tag{1.3}
  \]

  \medskip

  \begin{itemize}
    \setlength{\itemsep}{8pt}
    \item The bootstrap bias is a \alert{random variable in the limit} ---
      it tracks $B$ with irreducible noise $\xi_2$.
    \item Hence the bootstrap law of $\Tns$, conditional on $\Dn$, is
      \alert{random in the limit}: it cannot mimic the law of $\Tn$.
    \item $\pnh := \Pstar(\Tns \le \Tn)$ is \textbf{not asymptotically uniform}.
    \item Consequence: tests/CIs do not deliver nominal level
      (a nominal $5\%$ test may reject at $\sim 15\%$).
  \end{itemize}

  \medskip

  \begin{alertblock}{Goal of the paper}
    Restore valid inference based on $\Tn$ when $B$ is not consistently
    estimable --- by working with the bootstrap $p$-value $\pnh$ directly,
    not by trying to estimate $B$.
  \end{alertblock}

\end{frame}

% ============================================================

\begin{frame}{A way out --- prepivoting}

  \textbf{Idea (Beran 1987, 1988):} let
  \[
    H(u) \;:=\; \lim_{n} \PR(\pnh \le u)
    \qquad\text{(asymptotic cdf of the bootstrap $p$-value).}
  \]
  Even though $H \neq U[0,1]$, it can be:
  \begin{itemize}
    \setlength{\itemsep}{4pt}
    \item \textbf{continuous}, and
    \item \textbf{free of the unknown bias $B$}.
  \end{itemize}

  \medskip

  If so, the probability integral transform gives
  \[
    H(\pnh) \;\dto\; U_{[0,1]}.
  \]
  Replace $\pnh$ by the modified $p$-value $\pnt := \Hnh(\pnh)$, where
  $\Hnh \pto H$ uniformly.

  \medskip

  \begin{itemize}
    \setlength{\itemsep}{4pt}
    \item Two implementations: \textbf{plug-in} (when $H = H_\gamma$ has finite-dim $\gamma$)
      and \textbf{double bootstrap} (estimate $H$ nonparametrically).
    \item Beran proposed this for \textit{refinements}; here the same device
      \alert{rescues a broken bootstrap}.
  \end{itemize}

\end{frame}

\section{Motivating Examples}
% ============================================================

\begin{frame}{Two motivating examples}

  Two concrete settings where (1.1)--(1.3) hold.

  \medskip

  For each one we ask:
  \begin{enumerate}
    \setlength{\itemsep}{6pt}
    \item What is the model and the parameter of interest?
    \item Does \textbf{(1.1)} hold? --- $\Tn \dto B + \xi_1$ with $B\neq 0$.
    \item Does \textbf{(1.2)} hold? --- the bootstrap reproduces the noise $\xi_1$.
    \item Does \textbf{(1.3)} hold? --- $\Bnh - B \dto \xi_2 \neq 0$.
      \alert{This is the condition that makes the standard bootstrap fail.}
  \end{enumerate}

  \bigskip

  \begin{block}{The two examples}
    \begin{enumerate}
      \setlength{\itemsep}{2pt}
      \item \textbf{Ridge regression} with shrinkage tuned to the local-to-zero scale.
      \item \textbf{Nonparametric regression} with MSE-optimal bandwidth.
    \end{enumerate}
    \footnotesize
    A third example --- inference after model averaging (Hansen 2007) --- is in
    the paper; it shares the same mechanism as ridge.
  \end{block}

\end{frame}

% ============================================================

\begin{frame}{Example 1: Ridge regression --- setup}

  Linear regression $y_t = \theta'x_t + \varepsilon_t$, test $H_0: g'\theta = r$
  \;($g\in\R^p$ fixed, scalar restriction).

  \medskip

  Ridge estimator with shrinkage $c_n$:
  \[
    \tilde\theta_n = \tilde S_{xx}^{-1} S_{xy},
    \qquad \tilde S_{xx} := S_{xx} + n^{-1} c_n I_p.
  \]

  \medskip

  \textbf{Local-to-zero setting:} $\theta = \delta\, n^{-1/2}$ and $c_n / n \to c_0 > 0$.

  \medskip

  \footnotesize
  Interesting regime: regressors carry modest information about $\theta$
  ($\theta\sim n^{-1/2}$); calibrating $c_n \sim n$ puts \alert{shrinkage bias and
  sampling noise at the same order} --- the regime where the bias matters.

\end{frame}

% ============================================================

\begin{frame}{Example 1: Why (1.1)--(1.3) hold}

  \textbf{(1.1) holds.}
  \[
    \Tn = n^{1/2}(g'\tilde\theta_n - r)
        \;\dto\; \underbrace{B}_{\propto\, c_0\,\delta}
        \;+\; \underbrace{\xi_1}_{\sim N(0,v^2)}
  \]
  Bias survives in the limit and depends on the \alert{inestimable} $\delta$.

  \medskip

  \textbf{(1.2) holds.} \;Pairs bootstrap, centred at OLS $\hat\theta_n$:
  $\;\Tns - \Bnh \dstarto_p N(0,v^2)$.

  \medskip

  \begin{alertblock}{(1.3) holds --- the bootstrap bias misses by $O_p(1)$}
    \[
      \Bnh - \Bn = -\tfrac{c_n}{n}\,g'\tilde S_{xx}^{-1}\,n^{1/2}(\hat\theta_n - \theta)
      \;\dto\; \xi_2 \neq 0.
    \]
    $c_n/n \to c_0 > 0$ and $n^{1/2}(\hat\theta_n - \theta) = O_p(1)$ \alert{not $o_p(1)$}
    $\;\Rightarrow\;$ product is $O_p(1)$, irreducible.
  \end{alertblock}

  \footnotesize
  Centre at OLS, not ridge, to preserve $E^*[x_t^*\varepsilon_t^*]=0$
  (GMM-bootstrap recentring).

\end{frame}

% ============================================================

\begin{frame}{Example 2: Nonparametric regression --- setup}

  Nonparametric regression with smooth $\beta(\cdot)$:
  \[
    y_t = \beta(x_t) + \varepsilon_t,
    \qquad x_t = t/n,\;\; \varepsilon_t \sim \text{iid}(0,\sigma^2).
  \]

  \medskip

  Nadaraya--Watson estimator at fixed $x \in (0,1)$ with kernel $K$:
  \[
    \hat\beta_h(x) = \frac{1}{nh}\sum_{t=1}^{n} K\!\bigl(\tfrac{x_t-x}{h}\bigr)\, y_t,
    \qquad h = c\, n^{-1/5}.
  \]

  \smallskip

  \footnotesize
  $h = c\,n^{-1/5}$ is the MSE-optimal rate: it balances squared bias against
  variance and gives the fastest rate of convergence. But at this $h$ the
  \alert{bias does not vanish faster than the noise}, so it shows up in the
  asymptotic distribution.

\end{frame}

% ============================================================

\begin{frame}{Example 2: Why (1.1)--(1.3) hold}

  \textbf{(1.1) holds.} Taylor expansion at $x$ gives
  \[
    \Tn = (nh)^{1/2}\bigl(\hat\beta_h(x) - \beta(x)\bigr)
        \;=\; \underbrace{\Bn}_{\propto\,\beta''(x)}
        + \underbrace{\xi_{1,n}}_{\dto N(0,v^2)}.
  \]
  Bias depends on the \alert{inestimable-in-practice} $\beta''(x)$.

  \medskip

  \textbf{(1.2) holds (parametric bootstrap).}
  $y_t^* = \hat\beta_h(x_t) + \varepsilon_t^*$, $\varepsilon_t^* \sim N(0,\hat\sigma_n^2)$;
  centring at $\hat\beta_h(x)$ gives $\Tns - \Bnh \dstarto_p N(0,v^2)$.

  \medskip

  \begin{alertblock}{(1.3) holds --- different mechanism, same conclusion}
    The bootstrap bias $\Bnh$ replaces $\beta(\cdot)$ by $\hat\beta_h(\cdot)$ ---
    it \alert{re-smooths an already-smoothed curve}. The relevant curvature
    error is
    \[
      \hat\beta_h''(x) - \beta''(x) = O_p\bigl((nh)^{-1/2}\bigr),
    \]
    \textbf{exactly the same order as the bias itself} at the MSE-optimal
    bandwidth. Hence $\Bnh - \Bn \dto \xi_2 \neq 0$.
  \end{alertblock}

  \footnotesize
  $\beta''(x)$ can be estimated --- but it requires a second bandwidth and is
  unreliable in finite samples (Calonico, Cattaneo, Farrell 2018).

\end{frame}

% ============================================================

\begin{frame}{Examples --- recap}

  \begin{center}
  \footnotesize
  \begin{tabular}{@{}lcc@{}}
    \toprule
    & Ridge & Nonparametric \\
    \midrule
    Rate $g(n)$    & $n^{1/2}$ & $(nh)^{1/2} \approx n^{2/5}$ \\
    Bias $B$ from  & shrinkage to zero & smoothing curvature \\
    Inestimable    & $\delta$ (local to zero) & $\beta''(x)$ (impractical) \\
    Bootstrap      & pairs (centred at OLS) & parametric \\
    Mechanism      & $\hat\theta_n$ only $n^{1/2}$-consistent & MSE-optimal $h$ \\
    \bottomrule
  \end{tabular}
  \end{center}

  \bigskip

  \begin{alertblock}{One-line takeaway}
    \centering
    Bias and noise live at the \textbf{same rate} $\Longleftrightarrow$ (1.3) $\Longleftrightarrow$ bootstrap fails.
  \end{alertblock}

  \begin{itemize}
    \item Generic pattern, not a quirk of one model.
  \end{itemize}

  \medskip

  Now: a \textbf{general theory} that handles both at once.

\end{frame}

\section{General Results}
% ============================================================

\begin{frame}{Recap of notation (paper, Section 3)}

  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item $\Dn$: the original sample.
      $\;\Dns$: a bootstrap sample drawn given $\Dn$.
    \item $\Tn = g(n)(\thn - \theta)$: the original statistic;
      $\;\Tns = g(n)(\thns - \thn)$ its bootstrap version.
    \item $\Bn$: deterministic asymptotic bias of $\Tn$ (limit $B$).
    \item $\Bnh$: $\Dn$-measurable \alert{implicit} bootstrap bias.
    \item $\Pstar(\cdot) = P(\cdot \mid \Dn)$: bootstrap probability measure.
  \end{itemize}

  \medskip

  \begin{block}{Modes of convergence (used throughout)}
    \footnotesize
    \begin{itemize}
      \setlength{\itemsep}{2pt}
      \item $Y_n \pto \xi$ \;: convergence in probability (under $P$).
      \item $Y_n \dto \xi$ \;: convergence in distribution (under $P$).
      \item $Y_n^* \pto_p 0$ \;: $\Pstar(|Y_n^*|>\varepsilon) \pto 0$ for all $\varepsilon > 0$.
      \item $Y_n^* \dstarto_p \xi$ \;: $\Pstar(Y_n^* \le u) \pto P(\xi \le u)$ at every continuity point of the cdf.
    \end{itemize}
  \end{block}

  \footnotesize
  $G$: cdf of $\xi_1$. \;$F$: cdf of $\xi_1 - \xi_2$. \;$H = F\circ G^{-1}$ will be the cdf of $\pnh$.

\end{frame}

% ============================================================

\begin{frame}{Assumption 1 --- the original statistic}

  \begin{assumption}[1]
    $\Tn - \Bn \dto \xi_1$, where $\xi_1$ is centred at zero with cdf
    \[
      G(u) := \PR(\xi_1 \le u)
    \]
    that is \alert{continuous and strictly increasing} on its support.
  \end{assumption}

  \bigskip

  \textbf{Why these conditions on $G$?}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \textbf{Continuous:} needed for Polya's theorem
      (pointwise $\Rightarrow$ uniform convergence of cdfs).
    \item \textbf{Strictly increasing:} ensures $G^{-1}$ is well-defined, which we need
      for the probability integral transform $G(\xi_1) \sim U[0,1]$.
    \item Not restricted to Gaussian --- covers stable distributions, etc.
  \end{itemize}

\end{frame}

% ============================================================

\begin{frame}{Assumption 2 --- the bootstrap side}

  \begin{assumption}[2]
    There exists $\Dn$-measurable $\Bnh$ such that:
    \begin{enumerate}
      \item[(i)] $\Tns - \Bnh \;\dstarto_p\; \xi_1$
      \item[(ii)] $\displaystyle \begin{pmatrix}\Tn - \Bn \\ \Bnh - \Bn\end{pmatrix}
                  \dto \begin{pmatrix}\xi_1 \\ \xi_2\end{pmatrix}$, \;with
                  $F(u) := \PR(\xi_1 - \xi_2 \le u)$ continuous.
    \end{enumerate}
  \end{assumption}

  \bigskip

  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item $\Bnh$ is \alert{not} a deliberate estimator of $\Bn$ --- it is whatever bias the
      bootstrap DGP \textit{implicitly generates}.
    \item Part (i): the bootstrap captures the noise correctly.
    \item Part (ii): joint convergence; \alert{$\xi_2$ quantifies the bias estimation failure}.
    \item $F$ must be continuous (but not necessarily strictly increasing) so that $H = F \circ G^{-1}$ is continuous.
  \end{itemize}

\end{frame}

% ============================================================

\begin{frame}{Benchmark: what if $\Bnh$ \textit{were} consistent?}

  Suppose $\Bnh - \Bn = o_p(1)$, i.e.\ $\xi_2 = 0$ a.s. Then $F = G$ and:

  \bigskip

  \begin{align*}
    \pnh &= \Pstar(\Tns \le \Tn)
         = \Pstar(\Tns - \Bnh \le \Tn - \Bnh) \\[4pt]
    &= G(\Tn - \Bnh) + o_p(1)
    \qquad\text{\footnotesize(Assumption 2(i) + Polya)} \\[4pt]
    &\dto G(\xi_1)
    \qquad\text{\footnotesize(since $\Tn - \Bnh = \Tn - \Bn + o_p(1) \dto \xi_1$)} \\[4pt]
    &\sim U[0,1]
    \qquad\text{\footnotesize(probability integral transform, using $F = G$)}
  \end{align*}


\end{frame}

% ============================================================

\begin{frame}{Theorem 3.1 --- the limit of $\pnh$}

  \begin{block}{Theorem 3.1 (Cavaliere et al.\ 2024)}
    Under Assumptions 1 and 2,
    \[
      \pnh \;\dto\; G\bigl(F^{-1}(U_{[0,1]})\bigr).
    \]
  \end{block}

  \bigskip

  \textbf{Implications:}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item $\pnh$ is \textbf{not uniform} unless $G = F$ (i.e.\ $\xi_2 = 0$).
    \item Cdf of the limit $V := G(F^{-1}(U))$:
      \[
        \PR(V\le u) = \PR\!\bigl(U \le F(G^{-1}(u))\bigr) = F(G^{-1}(u)) =: H(u).
      \]
    \item $H$ continuous and \alert{free of $B$} ($F,G$ are laws of centred objects).
  \end{itemize}

\end{frame}

% ============================================================

\begin{frame}{Theorem 3.1 --- proof}

  \textbf{Step 1.} Rewrite, subtracting $\Bnh$ (constant under $\Pstar$):
  \[
    \pnh = \Pstar(\Tns - \Bnh \le \Tn - \Bnh)
  \]
  By Assumption 2(i), $\Pstar(\Tns - \Bnh \le u) \to G(u)$ \textit{uniformly} (Polya). Hence:
  \[
    \pnh = G(\Tn - \Bnh) + o_p(1)
  \]

  \textbf{Step 2.} Decompose $\Tn - \Bnh = (\Tn - \Bn) - (\Bnh - \Bn)$.
  By Assumption 2(ii) and the continuous mapping theorem:
  \[
    \Tn - \Bnh \;\dto\; \xi_1 - \xi_2
  \]

  \textbf{Step 3.} By continuity of $G$ and continuous mapping:
  $\;G(\Tn - \Bnh) \dto G(\xi_1 - \xi_2)$.

  \medskip

  \textbf{Step 4.} Since $\xi_1 - \xi_2$ has continuous cdf $F$, the probability integral transform
  gives $\xi_1 - \xi_2 \overset{d}{=} F^{-1}(U)$. Substitute:
  \[
    \pnh \;\dto\; G(F^{-1}(U_{[0,1]})) \qquad\square
  \]

\end{frame}

% ============================================================

\begin{frame}{Three remarks}

  \begin{alertblock}{Remark 3.1 --- the most important}
    If a bootstrap scheme gives $\Bnh \pto 0$, then $\xi_2 = -B$ (constant)
    and $H$ \alert{depends on $B$} --- prepivoting cannot work.
    Lesson: prefer schemes that try to capture the bias (even noisily).
  \end{alertblock}

  \bigskip

  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \textbf{Remark 3.2.} Confidence intervals are similarly invalid:
      $P(\theta_0 \in CI^{1-\alpha}_n) = P(\pnh \le 1-\alpha) \neq 1-\alpha$.
    \item \textbf{Remark 3.3.} When $\xi_2$ is nondegenerate, the bootstrap conditional
      distribution $\Lnh$ is \textit{random in the limit}: $\Lnh(u) \to G(u - B - \xi_2)$.
  \end{itemize}

\end{frame}

% ============================================================

\begin{frame}{Corollary 3.1 --- prepivoting restores validity}


  \medskip

  \begin{block}{Corollary 3.1}
    Under the conditions of Theorem 3.1, $\;H(\pnh) \dto U_{[0,1]}$.
  \end{block}

  \medskip

  \textit{Proof.} Let $V := G(F^{-1}(U_{[0,1]}))$ be the limit from Theorem 3.1.
  Apply $H$:
  \[
    H(V) = F\bigl(G^{-1}(G(F^{-1}(U_{[0,1]})))\bigr) = F(F^{-1}(U_{[0,1]})) = U_{[0,1]}.
  \]
  By continuous mapping, $\;H(\pnh) \dto H(V) = U_{[0,1]}$. \quad$\square$

  \medskip

  \textbf{Reading:} applying $H$ ``unwarps'' the distorted distribution of $\pnh$
  back to uniform --- exactly what the standard bootstrap was supposed to deliver.

  \medskip

  \alert{But $H$ depends on the unknown $F,G$ $\Longrightarrow$ we need an estimator $\Hnh$.}

\end{frame}

% ============================================================

\begin{frame}{Approach A: Plug-in}

  When $H = H_\gamma$ for some finite-dimensional $\gamma$, estimate $\gamma$
  and plug in.

  \medskip

  \begin{block}{Corollary 3.2}
    If $H_\gamma(u)$ is continuous in $(\gamma,u)$ and $\hat\gamma_n \pto \gamma$, then
    \[
      \pnt := H_{\hat\gamma_n}(\pnh) \;\dto\; U_{[0,1]}.
    \]
  \end{block}

  \medskip

  \textit{Proof.} Two pieces.
  \begin{enumerate}
    \setlength{\itemsep}{4pt}
    \item By joint continuity of $H_\gamma$ in $\gamma$,
      $\;\sup_u |H_{\hat\gamma_n}(u) - H_\gamma(u)| \pto 0$.
      Hence $\pnt - H(\pnh) \pto 0$.
    \item By Corollary 3.1, $H(\pnh) \dto U_{[0,1]}$.
  \end{enumerate}
  Slutsky $\Rightarrow$ $\pnt \dto U_{[0,1]}$. \;$\square$

  \medskip

  \footnotesize
  Cheap (one bootstrap + one formula), but requires deriving $H_\gamma$ in
  closed form. In the Gaussian case --- which covers our examples --- it
  takes a simple shape: see next slide.

\end{frame}

\begin{frame}{Gaussian plug-in --- the correction $m$}

  $\xi_1 \sim N(0,v^2)$, $(\xi_1,\xi_2)$ jointly normal, $\xi_1-\xi_2 \sim N(0,v_d^2)$.
  
    \medskip

  \[
    G(u) = \Phi(u/v),\quad F(u) = \Phi(u/v_d),\quad G^{-1}(u) = v\,\Phi^{-1}(u).
  \]
  \[
    H(u) = F(G^{-1}(u))
         = \Phi\!\left(\tfrac{v\,\Phi^{-1}(u)}{v_d}\right)
         = \Phi\!\bigl(m^{-1}\Phi^{-1}(u)\bigr),
    \qquad m := v_d/v.
  \]

  \medskip
  That is:
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Under (1.3): $\Tn - \Bnh \dto N(0,v_d^2)$ --- the
      \alert{actual spread is $v_d^2 \neq v^2$}.
    \item $\pnt = \Phi(m^{-1}\Phi^{-1}(\pnh))$ rescales the $p$-value as if
      the spread had always been $v_d^2$. \alert{$B$ drops out completely.}
  \end{itemize}

  \medskip

  \begin{block}{In our examples}
    \footnotesize
    \textbf{Ridge:} $m^2$ depends on $\Sigma_{xx}, c_0$ ---
    \alert{not} on the unknown $\delta$.\\
    \textbf{Nonparametric:} $m^2$ depends \alert{only on the kernel $K$}
    ($\sigma^2$ cancels) --- no estimation at all.
  \end{block}

\end{frame}

\begin{frame}{Approach B: Double bootstrap}

  \medskip

  \textbf{Construction:}
  \begin{enumerate}
    \setlength{\itemsep}{4pt}
    \item From each bootstrap sample $\Dns$, draw a second-level sample $\Dnss$.
    \item Compute $\Tnss$ and the second-level $p$-value
      $\;\pnhs := \Pstst(\Tnss \le \Tns)$.
    \item Estimate $H$ by $\;\Hnh(u) := \Pstar(\pnhs \le u)$.
    \item Modified $p$-value: $\;\pnt := \Hnh(\pnh) = \Pstar(\pnhs \le \pnh)$.
  \end{enumerate}

  \bigskip

  \textbf{Assumption 3} ensures the double bootstrap replicates the distortion:
  \begin{itemize}
    \item[(i)] $\Tnss - \Bnhs \dstarstarto_{p^*} \xi_1$ (noise replicated at level 2)
    \item[(ii)] $\Tns - \Bnhs \dstarto_p \xi_1 - \xi_2$ (bias gap replicated)
  \end{itemize}

  \medskip

  \begin{block}{Theorem 3.2}
    Under Assumptions 1, 2, 3: $\;\pnt = \Hnh(\pnh) \dto U_{[0,1]}$.
  \end{block}


\end{frame}

% ============================================================

\begin{frame}{Corollary 3.3 --- the practical shortcut}

  \begin{block}{Corollary 3.3}
    Under Assumptions 1, 2, 3:
    \[
      \pnt = \Pstar(\Tns - \Bnhs \;\le\; \Tn - \Bnh) + o_p(1).
    \]
  \end{block}

  \bigskip

   the double bootstrap modified $p$-value is asymptotically equivalent to a
  \alert{single-level bootstrap} applied to the bias-corrected statistic $\Tn - \Bnh$.

  \bigskip

  \textbf{Practical recipe:}
  \begin{enumerate}
    \setlength{\itemsep}{4pt}
    \item Compute $\Bnh$ from the original data (available in closed form for ridge, model averaging).
    \item Compute $\Bnhs$ from each bootstrap sample.
    \item Bootstrap $p$-value: $\Pstar(\Tns - \Bnhs \le \Tn - \Bnh)$.
  \end{enumerate}

  \medskip

  No double bootstrap needed when $\Bnh$, $\Bnhs$ are analytically available.

  \medskip

  \footnotesize
  Note: $\Bnh$ does \textit{not} need to be consistent for $\Bn$ --- the method works
  regardless.

\end{frame}

\begin{frame}{Ridge regression --- implementing the fix}

  Lemma 4.4 verifies Assumptions 1--2 with $(\xi_1, \xi_2) \sim N(0,V)$,
  $V$ depending on $\sigma^2, \Sigma_{xx}, c_0$ (all estimable).
  The distortion simplifies to:
  \[
    m^2 = \frac{g'\Sigma_{xx}^{-1}g}
               {g'\tilde\Sigma_{xx}^{-1}\Sigma_{xx}\tilde\Sigma_{xx}^{-1}g}
  \]

  \textbf{Plug-in:}
  \begin{enumerate}
    \setlength{\itemsep}{6pt}
    \item Compute ridge $\tilde\theta_n$ and $\Tn = n^{1/2}(g'\tilde\theta_n - r)$.
    \item Run pairs bootstrap: resample $(y_t^*, x_t^*)$, centre at OLS:
      $T_n^{*(b)} = n^{1/2}g'(\tilde\theta_n^{*(b)} - \hat\theta_n)$.
      Compute $\pnh$.
    \item $\hat{m}_n^2 = (g'S_{xx}^{-1}g)\,/\,(g'\tilde{S}_{xx}^{-1}S_{xx}\tilde{S}_{xx}^{-1}g)$.
    \item Report $\pnt = \Phi(\hat{m}_n^{-1}\,\Phi^{-1}(\pnh))$.
  \end{enumerate}

  \medskip

  Lemma 4.5 verifies Assumption 3 (double bootstrap also valid).

  
\end{frame}

\begin{frame}{Nonparametric regression --- implementing the fix}

  Lemma 4.6 verifies Assumptions 1--2. Remarkable: $\sigma^2$ cancels in
  $m^2 = v_d^2/v^2$, so $m$ depends \alert{only on the kernel $K$} --- known
  to the researcher, no estimation needed.

  \bigskip

  \textbf{Plug in:}
  \begin{enumerate}
    \setlength{\itemsep}{6pt}
    \item Compute $\hat\beta_h(x)$ with MSE-optimal $h = cn^{-1/5}$
      and $\Tn = (nh)^{1/2}(\hat\beta_h(x) - \beta(x))$.
    \item Run parametric bootstrap: $y_t^* = \hat\beta_h(x_t) + \varepsilon_t^*$,
      $\varepsilon_t^* \sim N(0, \hat\sigma_n^2)$.
      Compute $\pnh$.
    \item Look up $m$ from $K$ (a pre-computed constant).
    \item Report $\pnt = \Phi(m^{-1}\,\Phi^{-1}(\pnh))$.
  \end{enumerate}

  \bigskip

  \textbf{Comparison:}
  \textit{undersmoothing} removes bias but kills local power;
  \textit{robust bias correction} (CCF) needs a second bandwidth.
  This paper: keep MSE-optimal $h$, one-line correction, no $\beta''(x)$.

  \medskip

  \footnotesize
  Lemma 4.7 verifies Assumption 3 for the double bootstrap (with a small
  observable correction to $\Tnss$).

\end{frame}

\section{Wrap-up}
% ============================================================

\begin{frame}{Summary}

  \begin{itemize}
    \setlength{\itemsep}{8pt}
    \item \textbf{Problem.} When the estimator has an asymptotic bias $B$ that the
      bootstrap cannot replicate (1.3), the bootstrap distribution is
      \alert{random in the limit} and $\pnh$ is not uniform.
    \item \textbf{Diagnosis.} Bias and noise live at the same rate; an inestimable
      quantity ($\delta$ in model averaging / ridge, $\beta''(x)$ in nonparametrics)
      sits exactly at the boundary.
    \item \textbf{Fix --- prepivoting.} The asymptotic cdf $H = F\circ G^{-1}$ of $\pnh$
      is continuous and \alert{free of $B$}; transform $\pnt = \Hnh(\pnh) \dto U_{[0,1]}$.
    \item \textbf{Two routes.} Plug-in (Gaussian: ratio of variances $m^2 = v_d^2/v^2$,
      no $B$) or double bootstrap (no closed form for $H$ needed; Corollary~3.3
      gives a single-bootstrap shortcut).
  \end{itemize}

  \medskip

  \begin{block}{Take-away for the reading group}
    A new, third role for the bootstrap: \textit{not} variance, \textit{not} refinement
    --- but \alert{rescuing inference} when the first-order bootstrap is invalid,
    without ever estimating the bias.
  \end{block}

\end{frame}

  \]
  \[
    H(u) = \Phi(m^{-1}\Phi^{-1}(u)), \quad m = v_d/v
  \]

  Modified $p$-value: $\pnt = \Phi(\hat{m}_n^{-1}\Phi^{-1}(\pnh))$.

  \medskip

  \footnotesize
  Covers Examples 1--3 and Example 5 (panel). Example 4 (heavy tails)
  uses $G(u) = \Psi_\alpha(\omega^{-1}u)$ with $\Psi_\alpha$ the stable cdf.

\end{frame}

\begin{frame}[noframenumbering]{Appendix: Section 3.5 --- random centering (pairs bootstrap)}

  Some bootstrap schemes (e.g.\ pairs in model averaging) give
  $\Tns - B_n^* \dstarto_p \xi_1$ where $B_n^*$ depends on bootstrap data.

  \medskip

  Assumption 4 replaces 2(i): $\Tns - \Bnh \dstarto_p \zeta_1$ with cdf $J \neq G$.

  \medskip

  Theorem 3.4: $\pnh \dto J(F^{-1}(U))$ and $H(u) = F(J^{-1}(u))$.

  \medskip

  Plug-in and double bootstrap go through with $J$ replacing $G$.

  \bigskip

  \alert{Caution:} the plug-in formula for $m$ is bootstrap-scheme-specific.
  If you derived $m$ for FRB and applied it to pairs, you'd get the wrong correction.
  The double bootstrap handles this automatically.

\end{frame}

\begin{frame}[noframenumbering]{Appendix: Why prepivoting, not studentisation?}

  \textbf{Q.} Why not just bootstrap the studentised statistic $T_n / \hat{se}_n$?

  \medskip

  \textbf{A.} Studentisation fixes \alert{scale}, not \alert{location}.

  \medskip

  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Distortion here is $\hat B_n - B_n \dto \xi_2$ --- a \textbf{random shift}.
    \item Dividing by $\hat{se}_n$ does not subtract a random location.
    \item Studentisation $\Rightarrow$ refinements when bias is $o_p(1)$;
      useless when bias is $O_p(1)$ relative to noise.
  \end{itemize}

\end{frame}

\begin{frame}[noframenumbering]{Appendix: Power (Theorem 3.3)}

  Test $H_0: \theta = \bar\theta$ vs.\ $H_1: \theta < \bar\theta$.

  \medskip

  Under local alternatives $\theta_0 = \bar\theta + a/g(n)$ with $a < 0$:
  \[
    P(H(\pnh(\bar\theta)) \le \alpha) \;\to\; F(F^{-1}(\alpha) - a)
  \]

  \begin{itemize}
    \item Fixed alternatives ($a \to -\infty$): test is consistent.
    \item Local alternatives: power depends on $F$ only, not $G$.
    \item Same power as an oracle test that knows $B$.
  \end{itemize}

\end{frame}

\begin{frame}[noframenumbering]{Appendix: Supplement examples}

  \textbf{Example 4 --- Heavy tails:} $y_t = \theta + \varepsilon_t$ with
  $\varepsilon_t \sim S(\alpha)$, $\alpha \in (1,2)$ (infinite variance).
  \begin{itemize}
    \item Non-Gaussian $\xi_1$, unknown convergence rate $n^{1-1/\alpha}$.
    \item Plug-in works: $H(u) = \Psi_\alpha(\omega\Psi_\alpha^{-1}(u))$.
    \item $m$-out-of-$n$ bootstrap fails (zeros out bias $\Rightarrow$ Remark 3.1 applies).
  \end{itemize}

  \bigskip

  \textbf{Example 5 --- Dynamic panel:} Fixed effects MLE with incidental parameter bias.
  \begin{itemize}
    \item Cross-sectional pairs bootstrap gives $\Bnh = 0$
      $\;\Rightarrow\;$ distortion depends on $B$ (Remark 3.1).
    \item But $B$ \textit{can} be estimated consistently here (Hahn--Kuersteiner 2011).
    \item Plug-in works; \alert{double bootstrap fails}
      (Remark B.4: both levels zero out the bias).
  \end{itemize}

\end{frame}
